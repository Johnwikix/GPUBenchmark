add_executable(opencl_worker main.cpp)
target_link_libraries(opencl_worker PRIVATE gpubench_common dxgi)
target_include_directories(opencl_worker PRIVATE ${PROJECT_SOURCE_DIR}/third_party/OpenCL-Headers)

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/kernels/opencl/alu.cl)
