# CUDA 13.2 supports the installed MSVC 195x host and Blackwell GeForce.
find_package(CUDAToolkit 13.2 REQUIRED)
if(NOT CUDAToolkit_VERSION VERSION_LESS 13.3)
  message(FATAL_ERROR "The locked CUDA backend requires CUDA 13.2.x; audit and relock a newer toolkit before using it")
endif()
option(GPUBENCH_CUDA_DIAGNOSTICS "Build isolated precision/layout performance diagnostics" OFF)
if(GPUBENCH_CUDA_DIAGNOSTICS)
  add_executable(cuda_gemm_probe ${PROJECT_SOURCE_DIR}/tests/diagnostics/cuda_gemm_probe.cpp kernels.cu)
  # Localized MSVC /showIncludes is not reliably parsed by installed Ninja.
  # This diagnostic includes production .cpp; explicitly track both inputs.
  set_property(SOURCE ${PROJECT_SOURCE_DIR}/tests/diagnostics/cuda_gemm_probe.cpp APPEND PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp;${PROJECT_SOURCE_DIR}/tests/diagnostics/extended_search.hpp")
  target_link_libraries(cuda_gemm_probe PRIVATE gpubench_common CUDA::cudart_static)
  target_include_directories(cuda_gemm_probe PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
  target_compile_definitions(cuda_gemm_probe PRIVATE GPUBENCH_CUDA_BASELINE="${CUDAToolkit_VERSION}")
  set_target_properties(cuda_gemm_probe PROPERTIES CUDA_ARCHITECTURES "120-real" CUDA_STANDARD 17 CUDA_RUNTIME_LIBRARY Static)
  target_compile_options(cuda_gemm_probe PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=true;-Xcompiler=/MT;-Xcompiler=/utf-8;-Xcompiler=/EHsc>)
  if(GPUBENCH_CUPTI_ROOT)
    target_include_directories(cuda_gemm_probe PRIVATE "${GPUBENCH_CUPTI_ROOT}/include")
    find_library(GPUBENCH_DIAG_CUPTI_LIBRARY NAMES cupti PATHS "${GPUBENCH_CUPTI_ROOT}/lib" "${GPUBENCH_CUPTI_ROOT}/lib/x64" NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(cuda_gemm_probe PRIVATE "${GPUBENCH_DIAG_CUPTI_LIBRARY}")
    target_compile_definitions(cuda_gemm_probe PRIVATE GPUBENCH_DIAG_CUPTI=1)
  endif()
endif()
if(MSVC_VERSION GREATER_EQUAL 1960)
  message(FATAL_ERROR "CUDA 13.2 supports MSVC through 195x. Select a supported MSVC toolset; unsupported-compiler overrides are prohibited")
endif()
enable_language(CUDA)
add_executable(cuda_worker main.cpp kernels.cu)
set(GPUBENCH_CUTLASS_ROOT "" CACHE PATH "Optional pinned CUTLASS v4.3.4 root for native SM120 FP8 GEMM")
if(GPUBENCH_CUTLASS_ROOT)
  file(READ "${GPUBENCH_CUTLASS_ROOT}/include/cutlass/version.h" GPUBENCH_CUTLASS_VERSION)
  if(NOT GPUBENCH_CUTLASS_VERSION MATCHES "CUTLASS_MAJOR 4" OR NOT GPUBENCH_CUTLASS_VERSION MATCHES "CUTLASS_MINOR 3" OR NOT GPUBENCH_CUTLASS_VERSION MATCHES "CUTLASS_PATCH 4")
    message(FATAL_ERROR "Native SM120 path requires pinned CUTLASS 4.3.4")
  endif()
  add_library(gpubench_sm120 STATIC sm120_gemm.cu)
  target_link_libraries(gpubench_sm120 PUBLIC CUDA::cuda_driver)
  target_include_directories(gpubench_sm120 PRIVATE "${GPUBENCH_CUTLASS_ROOT}/include" "${GPUBENCH_CUTLASS_ROOT}/tools/util/include")
  set_target_properties(gpubench_sm120 PROPERTIES CUDA_ARCHITECTURES "120a-real" CUDA_STANDARD 17 CUDA_RUNTIME_LIBRARY Static)
  target_compile_options(gpubench_sm120 PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr;-Xcompiler=/MT;-Xcompiler=/utf-8;-Xcompiler=/EHsc;-Xcompiler=/Zc:preprocessor;-Xcompiler=/Zc:__cplusplus>)
  target_link_libraries(cuda_worker PRIVATE gpubench_sm120)
  target_compile_definitions(cuda_worker PRIVATE GPUBENCH_NATIVE_SM120=1)
  add_executable(sm120_kernel_check ${PROJECT_SOURCE_DIR}/tests/diagnostics/sm120_kernel_check.cpp)
  target_include_directories(sm120_kernel_check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(sm120_kernel_check PRIVATE gpubench_sm120 CUDA::cudart_static)
  if(TARGET cuda_gemm_probe)
    target_link_libraries(cuda_gemm_probe PRIVATE gpubench_sm120)
    target_compile_definitions(cuda_gemm_probe PRIVATE GPUBENCH_NATIVE_SM120=1)
  endif()
endif()
target_link_libraries(cuda_worker PRIVATE gpubench_common CUDA::cudart_static)
target_include_directories(cuda_worker PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
set_target_properties(cuda_worker PROPERTIES CUDA_ARCHITECTURES "75-real;86-real;89-real;120-real;120-virtual" CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES CUDA_RUNTIME_LIBRARY Static)
target_compile_options(cuda_worker PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=true;-Xptxas=-v;-Xcompiler=/MT;-Xcompiler=/utf-8;-Xcompiler=/EHsc>)
target_compile_definitions(cuda_worker PRIVATE GPUBENCH_CUDA_BASELINE="${CUDAToolkit_VERSION}")
# cuBLASLt is explicitly loaded from the worker's directory. It is intentionally
# absent from the PE import table, allowing ALU/enumeration when BLAS is missing.
install(TARGETS cuda_worker RUNTIME DESTINATION workers/cuda)
install(FILES README.md REFERENCES.md DESTINATION workers/cuda)
find_file(GPUBENCH_CUBLASLT_RUNTIME cublasLt64_13.dll PATHS "${CUDAToolkit_BIN_DIR}/x64" "${CUDAToolkit_BIN_DIR}" NO_DEFAULT_PATH)
if(GPUBENCH_CUBLASLT_RUNTIME)
  install(FILES "${GPUBENCH_CUBLASLT_RUNTIME}" DESTINATION workers/cuda)
else()
  message(WARNING "cuBLASLt DLL not found in the selected toolkit; ALU can run, GEMM requires the matching private runtime")
endif()
