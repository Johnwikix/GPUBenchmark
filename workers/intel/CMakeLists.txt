if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
  message(FATAL_ERROR "Intel worker requires the pinned Intel oneAPI DPC++ compiler (icx -fsycl), in its own build directory")
endif()
if(NOT CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 2026.0 OR
   NOT CMAKE_CXX_COMPILER_VERSION VERSION_LESS 2027.0)
  message(FATAL_ERROR "This source targets the oneAPI 2026 compiler/API series; update and validate the dependency lock before changing series")
endif()
set(MKL_INTERFACE lp64)
set(MKL_LINK dynamic)
set(MKL_THREADING sequential)
find_package(MKL 2026.0 CONFIG REQUIRED HINTS "$ENV{MKLROOT}")
add_executable(intel_worker intel_worker.cpp)
# Intel changed the imported target name across releases. Both refer to SYCL,
# never silently substitute the C/Fortran MKL::MKL target.
if(TARGET MKL::MKL_SYCL)
  target_link_libraries(intel_worker PRIVATE MKL::MKL_SYCL)
elseif(TARGET MKL::MKL_DPCPP)
  target_link_libraries(intel_worker PRIVATE MKL::MKL_DPCPP)
else()
  message(FATAL_ERROR "oneMKL package does not expose a supported SYCL target")
endif()
target_link_libraries(intel_worker PRIVATE gpubench_common dxgi)
target_compile_options(intel_worker PRIVATE -fsycl -fno-fast-math -ffp-contract=on -fsycl-device-code-split=per_kernel)
target_link_options(intel_worker PRIVATE -fsycl)
set_property(TARGET intel_worker PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
install(TARGETS intel_worker RUNTIME DESTINATION workers/intel)
install(FILES README.md SOURCES.json DESTINATION workers/intel)
