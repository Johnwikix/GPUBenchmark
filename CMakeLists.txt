cmake_minimum_required(VERSION 3.25)
project(gpubench VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "gpubench requires native Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(GPUBENCH_BACKEND host CACHE STRING "host|cuda|amd|intel|opencl; one build directory per backend")
add_library(gpubench_common INTERFACE)
target_include_directories(gpubench_common INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src/common ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
target_compile_definitions(gpubench_common INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN GPUBENCH_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(gpubench_common INTERFACE $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
endif()

# Explicit project-header dependencies also work when localized MSVC
# /showIncludes diagnostics cannot be decoded by the installed Ninja/CMake.
# SDK headers belong to the immutable version-locked toolchain.
file(GLOB_RECURSE GPUBENCH_HEADERS CONFIGURE_DEPENDS
  "${PROJECT_SOURCE_DIR}/src/*.hpp" "${PROJECT_SOURCE_DIR}/workers/*.hpp"
  "${PROJECT_SOURCE_DIR}/third_party/*.hpp" "${PROJECT_SOURCE_DIR}/third_party/*.h")
function(gpubench_header_dependencies directory)
  get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
  foreach(target IN LISTS targets)
    get_target_property(sources ${target} SOURCES)
    get_target_property(source_dir ${target} SOURCE_DIR)
    foreach(source IN LISTS sources)
      if(source MATCHES "\\.(cpp|cxx|cc|cu)$")
        get_filename_component(absolute "${source}" ABSOLUTE BASE_DIR "${source_dir}")
        set_property(SOURCE "${absolute}" DIRECTORY "${directory}" APPEND PROPERTY OBJECT_DEPENDS "${GPUBENCH_HEADERS}")
      endif()
    endforeach()
  endforeach()
  get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
  foreach(child IN LISTS children)
    gpubench_header_dependencies("${child}")
  endforeach()
endfunction()
if(GPUBENCH_BACKEND STREQUAL "host")
  add_executable(gpubench src/host/main.cpp)
  target_link_libraries(gpubench PRIVATE gpubench_common)
  enable_testing()
  add_executable(common_tests tests/unit/common_tests.cpp)
  target_link_libraries(common_tests PRIVATE gpubench_common)
  add_test(NAME common COMMAND common_tests)
  add_executable(protocol_fixture tests/integration/protocol_fixture.cpp)
  target_link_libraries(protocol_fixture PRIVATE gpubench_common)
  install(TARGETS gpubench RUNTIME DESTINATION .)
elseif(GPUBENCH_BACKEND MATCHES "^(cuda|amd|intel|opencl)$")
  add_subdirectory(workers/${GPUBENCH_BACKEND})
else()
  message(FATAL_ERROR "Unknown GPUBENCH_BACKEND")
endif()
gpubench_header_dependencies("${PROJECT_SOURCE_DIR}")
